cmake_minimum_required(VERSION 3.24)
project(ACUnityFirstPerson VERSION 0.8.5 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Use 64-bit Windows and the Visual Studio C++ compiler.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(imgui STATIC third_party/imgui/imgui.cpp third_party/imgui/imgui_draw.cpp
  third_party/imgui/imgui_tables.cpp third_party/imgui/imgui_widgets.cpp)
target_include_directories(imgui PUBLIC third_party/imgui)
add_library(ACUnityFirstPerson SHARED src/plugin.cpp)
target_compile_definitions(ACUnityFirstPerson PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(ACUnityFirstPerson PRIVATE /W4 /WX /permissive- /EHsc)
target_link_libraries(ACUnityFirstPerson PRIVATE user32 imgui)
include(CTest)
if(BUILD_TESTING)
  add_executable(camera_tests tests/camera_tests.cpp)
  target_include_directories(camera_tests PRIVATE src)
  add_test(NAME camera_geometry COMMAND camera_tests)
endif()

if(BUILD_TESTING)
  add_executable(plugin_contract_tests tests/plugin_contract_tests.cpp)
  target_include_directories(plugin_contract_tests PRIVATE src)
  target_link_libraries(plugin_contract_tests PRIVATE imgui)
  target_compile_definitions(plugin_contract_tests PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
  add_test(NAME plugin_contract COMMAND plugin_contract_tests $<TARGET_FILE:ACUnityFirstPerson>)
endif()
if(BUILD_TESTING)
  add_executable(head_anchor_tests tests/head_anchor_tests.cpp)

endif()
